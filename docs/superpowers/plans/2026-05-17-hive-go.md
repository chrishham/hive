# Hive-Go Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rewrite hive as a tmux-free Go application with embedded PTY terminals using Bubbletea + bubbleterm.

**Architecture:** Single Go binary running a Bubbletea TUI. Claude Code sessions run in PTYs managed by creack/pty, rendered inline via bubbleterm. Dashboard mode shows session list + synopsis; focus mode gives full-screen terminal access. A separate hive-hook binary handles Claude Code hook events. State files are compatible with the existing Python hive.

**Tech Stack:** Go 1.22+, Bubbletea (TUI), bubbleterm (terminal widget), creack/pty (PTY), Lip Gloss (styling), BurntSushi/toml (config), anthropic-sdk-go (synopsis API)

```go
package safety

import (
	"fmt"
	"os"
	"path/filepath"
	"regexp"
	"strings"
)

var (
	sessionNameRE = regexp.MustCompile(`^[A-Za-z0-9._-]{1,64}$`)
	replaceRE     = regexp.MustCompile(`[^A-Za-z0-9._-]`)
)

const (
	maxNameLen  = 64
	defaultName = "session"
)

type InvalidSessionName struct {
	Name string
}

func (e *InvalidSessionName) Error() string {
	return fmt.Sprintf("invalid session name: %q", e.Name)
}

func ValidateSessionName(name string) error {
	if !sessionNameRE.MatchString(name) {
		return &InvalidSessionName{Name: name}
	}
	if name == "." || name == ".." {
		return &InvalidSessionName{Name: name}
	}
	return nil
}

func SanitizeSessionName(name string) string {
	cleaned := replaceRE.ReplaceAllString(name, "_")
	if len(cleaned) > maxNameLen {
		cleaned = cleaned[:maxNameLen]
	}
	if cleaned == "" || cleaned == "." || cleaned == ".." || strings.Trim(cleaned, "_") == "" {
		return defaultName
	}
	return cleaned
}

func AtomicWriteText(path string, content string) error {
	info, err := os.Lstat(path)
	if err == nil {
		if info.Mode()&os.ModeSymlink != 0 {
			return fmt.Errorf("refusing to write through symlink: %s", path)
		}
	}

	dir := filepath.Dir(path)
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return fmt.Errorf("create parent dir: %w", err)
	}

	tmp, err := os.CreateTemp(dir, "."+filepath.Base(path)+".*")
	if err != nil {
		return fmt.Errorf("create temp file: %w", err)
	}
	tmpName := tmp.Name()

	_, writeErr := tmp.WriteString(content)
	syncErr := tmp.Sync()
	closeErr := tmp.Close()

	if writeErr != nil || syncErr != nil || closeErr != nil {
		os.Remove(tmpName)
		if writeErr != nil {
			return writeErr
		}
		if syncErr != nil {
			return syncErr
		}
		return closeErr
	}

	if err := os.Rename(tmpName, path); err != nil {
		os.Remove(tmpName)
		return fmt.Errorf("atomic rename: %w", err)
	}
	return nil
}
```

```go
package config

import (
	"os"
	"path/filepath"

	"github.com/BurntSushi/toml"
	"github.com/chrishham/hive-go/internal/safety"
)

type HiveConfig struct {
	ScanPaths         []string
	ClonePath         string
	RefreshIntervalMs int
	PreviewLines      int
	SessionName       string
}

type tomlFile struct {
	Projects struct {
		ScanPaths []string `toml:"scan_paths"`
		ClonePath string   `toml:"clone_path"`
	} `toml:"projects"`
	Display struct {
		RefreshIntervalMs int `toml:"refresh_interval_ms"`
		PreviewLines      int `toml:"preview_lines"`
	} `toml:"display"`
	Tmux struct {
		SessionName string `toml:"session_name"`
	} `toml:"tmux"`
}

func defaultScanPaths() []string {
	return []string{filepath.Join(os.Getenv("HOME"), "Projects")}
}

func defaultClonePath() string {
	return filepath.Join(os.Getenv("HOME"), "Projects")
}

func Defaults() HiveConfig {
	return HiveConfig{
		ScanPaths:         defaultScanPaths(),
		ClonePath:         defaultClonePath(),
		RefreshIntervalMs: 5000,
		PreviewLines:      20,
		SessionName:       "hive",
	}
}

func Load(path string) (HiveConfig, error) {
	c := Defaults()

	if _, err := os.Stat(path); os.IsNotExist(err) {
		return c, nil
	}

	var f tomlFile
	if _, err := toml.DecodeFile(path, &f); err != nil {
		return c, nil
	}

	if len(f.Projects.ScanPaths) > 0 {
		c.ScanPaths = f.Projects.ScanPaths
	}
	if f.Projects.ClonePath != "" {
		c.ClonePath = f.Projects.ClonePath
	}
	if f.Display.RefreshIntervalMs > 0 {
		c.RefreshIntervalMs = f.Display.RefreshIntervalMs
	}
	if f.Display.PreviewLines > 0 {
		c.PreviewLines = f.Display.PreviewLines
	}
	if f.Tmux.SessionName != "" {
		c.SessionName = safety.SanitizeSessionName(f.Tmux.SessionName)
	}

	return c, nil
}

func LoadDefault() (HiveConfig, error) {
	home, _ := os.UserHomeDir()
	return Load(filepath.Join(home, ".config", "hive", "config.toml"))
}
```

```go
package config

import (
	"encoding/json"
	"os"
	"path/filepath"
	"time"

	"github.com/chrishham/hive-go/internal/safety"
)

type SessionInfo struct {
	ProjectPath    string `json:"project_path"`
	CreatedAt      string `json:"created_at"`
	LastUsed       string `json:"last_used"`
	PID            int    `json:"pid"`
	ClaudeSessionID string `json:"claude_session_id"`
}

type ProjectInfo struct {
	LastUsed string `json:"last_used"`
}

type HiveState struct {
	Sessions map[string]SessionInfo `json:"sessions"`
	Projects map[string]ProjectInfo `json:"projects"`
}

func LoadState(path string) (*HiveState, error) {
	s := &HiveState{
		Sessions: make(map[string]SessionInfo),
		Projects: make(map[string]ProjectInfo),
	}

	data, err := os.ReadFile(path)
	if err != nil {
		if os.IsNotExist(err) {
			return s, nil
		}
		return s, nil
	}

	if err := json.Unmarshal(data, s); err != nil {
		return s, nil
	}
	if s.Sessions == nil {
		s.Sessions = make(map[string]SessionInfo)
	}
	if s.Projects == nil {
		s.Projects = make(map[string]ProjectInfo)
	}
	return s, nil
}

func LoadStateDefault() (*HiveState, error) {
	home, _ := os.UserHomeDir()
	return LoadState(filepath.Join(home, ".config", "hive", "state.json"))
}

func (s *HiveState) Save(path string) error {
	data, err := json.MarshalIndent(s, "", "  ")
	if err != nil {
		return err
	}
	return safety.AtomicWriteText(path, string(data))
}

func (s *HiveState) SaveDefault() error {
	home, _ := os.UserHomeDir()
	return s.Save(filepath.Join(home, ".config", "hive", "state.json"))
}

func (s *HiveState) AddSession(name, projectPath string, pid int, claudeSessionID string) {
	now := time.Now().UTC().Format(time.RFC3339)
	s.Sessions[name] = SessionInfo{
		ProjectPath:    projectPath,
		CreatedAt:      now,
		LastUsed:       now,
		PID:            pid,
		ClaudeSessionID: claudeSessionID,
	}
	s.Projects[projectPath] = ProjectInfo{LastUsed: now}
}

func (s *HiveState) RemoveSession(name string) {
	delete(s.Sessions, name)
}
```

```go
package main

import (
	"encoding/json"
	"io"
	"os"
	"time"

	"github.com/chrishham/hive-go/internal/safety"
)

const maxStdinBytes = 64 * 1024
const maxSessionIDLen = 256

var eventToState = map[string]string{
	"UserPromptSubmit": "working",
	"Stop":             "waiting",
	"Notification":     "waiting",
	"SessionStart":     "waiting",
}

type hookPayload struct {
	HookEventName string `json:"hook_event_name"`
	SessionID     string `json:"session_id"`
}

type stateRecord struct {
	State     string `json:"state"`
	SessionID string `json:"session_id"`
	UpdatedAt string `json:"updated_at"`
	Event     string `json:"event"`
}

func run() int {
	sessionName := os.Getenv("HIVE_SESSION")
	if sessionName == "" {
		return 0
	}

	raw, err := io.ReadAll(io.LimitReader(os.Stdin, maxStdinBytes+1))
	if err != nil || len(raw) > maxStdinBytes {
		return 0
	}

	var payload hookPayload
	if err := json.Unmarshal(raw, &payload); err != nil {
		return 0
	}

	home, err := os.UserHomeDir()
	if err != nil {
		return 0
	}

	stateDir := home + "/.claude/hive/state"
	statePath := stateDir + "/" + sessionName + ".json"

	if err := safety.ValidateSessionName(sessionName); err != nil {
		return 0
	}

	if payload.HookEventName == "SessionEnd" {
		os.Remove(statePath)
		return 0
	}

	state, ok := eventToState[payload.HookEventName]
	if !ok {
		return 0
	}

	sessionID := payload.SessionID
	if len(sessionID) > maxSessionIDLen {
		sessionID = sessionID[:maxSessionIDLen]
	}

	record := stateRecord{
		State:     state,
		SessionID: sessionID,
		UpdatedAt: time.Now().UTC().Format(time.RFC3339),
		Event:     payload.HookEventName,
	}

	data, err := json.Marshal(record)
	if err != nil {
		return 0
	}

	os.MkdirAll(stateDir, 0o755)
	if err := safety.AtomicWriteText(statePath, string(data)); err != nil {
		return 0
	}
	return 0
}

func main() {
	os.Exit(run())
}
```

```go
package hooks

import (
	"encoding/json"
	"os"
	"path/filepath"

	"github.com/chrishham/hive-go/internal/safety"
)

const HookCommand = `command -v hive-hook >/dev/null 2>&1 && hive-hook || true`

var HookEvents = []string{
	"UserPromptSubmit",
	"Stop",
	"SubagentStop",
	"Notification",
	"SessionStart",
	"SessionEnd",
}

var legacyCommands = []string{"hive-hook", "-m hive.hook_writer"}

func DefaultSettingsPath() string {
	home, _ := os.UserHomeDir()
	return filepath.Join(home, ".claude", "settings.json")
}

func isHiveEntry(entry map[string]interface{}) bool {
	hooks, ok := entry["hooks"].([]interface{})
	if !ok {
		return false
	}
	for _, h := range hooks {
		hm, ok := h.(map[string]interface{})
		if !ok {
			continue
		}
		cmd, _ := hm["command"].(string)
		if cmd == HookCommand {
			return true
		}
		for _, legacy := range legacyCommands {
			if len(cmd) >= len(legacy) && cmd[len(cmd)-len(legacy):] == legacy {
				return true
			}
		}
	}
	return false
}

func loadSettings(path string) (map[string]interface{}, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		if os.IsNotExist(err) {
			return make(map[string]interface{}), nil
		}
		return nil, err
	}
	var settings map[string]interface{}
	if err := json.Unmarshal(data, &settings); err != nil {
		return nil, err
	}
	return settings, nil
}

func Install(path string) error {
	dir := filepath.Dir(path)
	os.MkdirAll(dir, 0o755)

	settings, err := loadSettings(path)
	if err != nil {
		return err
	}

	hooks, ok := settings["hooks"].(map[string]interface{})
	if !ok {
		hooks = make(map[string]interface{})
		settings["hooks"] = hooks
	}

	for _, event := range HookEvents {
		var entries []interface{}
		if existing, ok := hooks[event].([]interface{}); ok {
			for _, e := range existing {
				em, ok := e.(map[string]interface{})
				if ok && isHiveEntry(em) {
					continue
				}
				entries = append(entries, e)
			}
		}

		newEntry := map[string]interface{}{
			"matcher": "",
			"hooks": []interface{}{
				map[string]interface{}{
					"type":    "command",
					"command": HookCommand,
				},
			},
		}
		entries = append(entries, newEntry)
		hooks[event] = entries
	}

	data, err := json.MarshalIndent(settings, "", "  ")
	if err != nil {
		return err
	}
	return safety.AtomicWriteText(path, string(data))
}

func Uninstall(path string) error {
	settings, err := loadSettings(path)
	if err != nil {
		return err
	}

	hooks, ok := settings["hooks"].(map[string]interface{})
	if !ok {
		return nil
	}

	for event := range hooks {
		entries, ok := hooks[event].([]interface{})
		if !ok {
			continue
		}
		var filtered []interface{}
		for _, e := range entries {
			em, ok := e.(map[string]interface{})
			if ok && isHiveEntry(em) {
				continue
			}
			filtered = append(filtered, e)
		}
		if len(filtered) == 0 {
			delete(hooks, event)
		} else {
			hooks[event] = filtered
		}
	}

	if len(hooks) == 0 {
		delete(settings, "hooks")
	}

	data, err := json.MarshalIndent(settings, "", "  ")
	if err != nil {
		return err
	}
	return safety.AtomicWriteText(path, string(data))
}

func IsInstalled(path string) bool {
	settings, err := loadSettings(path)
	if err != nil {
		return false
	}
	hooks, ok := settings["hooks"].(map[string]interface{})
	if !ok {
		return false
	}
	for _, entries := range hooks {
		arr, ok := entries.([]interface{})
		if !ok {
			continue
		}
		for _, e := range arr {
			em, ok := e.(map[string]interface{})
			if !ok {
				continue
			}
			hs, ok := em["hooks"].([]interface{})
			if !ok {
				continue
			}
			for _, h := range hs {
				hm, ok := h.(map[string]interface{})
				if !ok {
					continue
				}
				if hm["command"] == HookCommand {
					return true
				}
			}
		}
	}
	return false
}
```

```go
package detector

import (
	"regexp"
	"strconv"
	"strings"
)

const (
	StateWaiting       = "waiting"
	StateWorking       = "working"
	StateBootstrapping = "bootstrapping"
)

var promptPatterns = []*regexp.Regexp{
	regexp.MustCompile(`[❯>]\s*$`),
	regexp.MustCompile(`(?s)╭.*╮.*│.*>.*│.*╰.*╯`),
	regexp.MustCompile(`│\s*>\s*│`),
	regexp.MustCompile(`❯❯\s+(bypass permissions|plan|default)`),
	regexp.MustCompile(`\(shift\+tab to cycle\)`),
	regexp.MustCompile(`Enter to confirm`),
	regexp.MustCompile(`Do you want to proceed\?`),
	regexp.MustCompile(`(?i)\(Y\)es\b.*\(N\)o\b`),
}

var workingPatterns = []*regexp.Regexp{
	regexp.MustCompile(`[⠋⠙⠹⠸⠼⠴⠦⠧⠇⠏]`),
	regexp.MustCompile(`\(\d+s\s*·`),
	regexp.MustCompile(`esc to interrupt`),
	regexp.MustCompile(`Thinking\.\.\.`),
	regexp.MustCompile(`Running:`),
	regexp.MustCompile(`Reading:`),
	regexp.MustCompile(`Editing:`),
	regexp.MustCompile(`Writing:`),
}

var loadedPatterns = []*regexp.Regexp{
	regexp.MustCompile(`Claude Code`),
	regexp.MustCompile(`Welcome back`),
	regexp.MustCompile(`Tips for getting started`),
}

var modelPattern = regexp.MustCompile(`(?i)(Opus|Sonnet|Haiku)\s+(\d+\.\d+)(?:\s+\((\d+[KMG])\s+context\))?`)
var urlPattern = regexp.MustCompile(`(?:https?://)?(?:localhost|127\.0\.0\.1):(\d{2,5})`)
var contextPctPattern = regexp.MustCompile(`ctx:[█░]+ (\d+)%`)

func DetectState(paneText string) string {
	trimmed := strings.TrimSpace(paneText)
	if trimmed == "" {
		return StateBootstrapping
	}

	loaded := false
	for _, p := range loadedPatterns {
		if p.MatchString(paneText) {
			loaded = true
			break
		}
	}
	if !loaded {
		return StateBootstrapping
	}

	end := paneText
	if len(end) > 500 {
		end = end[len(end)-500:]
	}

	for _, p := range workingPatterns {
		if p.MatchString(end) {
			return StateWorking
		}
	}

	for _, p := range promptPatterns {
		if p.MatchString(end) {
			return StateWaiting
		}
	}

	return StateWorking
}

func DetectModel(paneText string) (model, context string) {
	m := modelPattern.FindStringSubmatch(paneText)
	if m == nil {
		return "", ""
	}
	name := strings.ToLower(m[1])
	version := m[2]
	ctx := ""
	if len(m) > 3 {
		ctx = m[3]
	}
	return name + "-" + version, ctx
}

func DetectURLs(scrollback string) []string {
	matches := urlPattern.FindAllStringSubmatch(scrollback, -1)
	seen := make(map[string]bool)
	var urls []string
	for _, m := range matches {
		key := "localhost:" + m[1]
		if !seen[key] {
			seen[key] = true
			urls = append(urls, key)
		}
	}
	return urls
}

func DetectContextPct(paneText string) int {
	end := paneText
	if len(end) > 500 {
		end = end[len(end)-500:]
	}
	m := contextPctPattern.FindStringSubmatch(end)
	if m == nil {
		return -1
	}
	pct, err := strconv.Atoi(m[1])
	if err != nil {
		return -1
	}
	if pct > 100 {
		return 100
	}
	return pct
}
```

```go
package process

import (
	"fmt"
	"os"
	"os/exec"
	"sync"
	"syscall"
	"time"

	"github.com/creack/pty"
)

type Session struct {
	Name       string
	ProjectDir string
	Cmd        *exec.Cmd
	PTY        *os.File
	PID        int
	ExitCh     chan struct{}
}

type Manager struct {
	mu       sync.Mutex
	sessions map[string]*Session
}

func NewManager() *Manager {
	return &Manager{
		sessions: make(map[string]*Session),
	}
}

func (m *Manager) Spawn(name, projectDir string, args []string, width, height int) (*Session, error) {
	m.mu.Lock()
	defer m.mu.Unlock()

	if _, exists := m.sessions[name]; exists {
		return nil, fmt.Errorf("session %q already exists", name)
	}

	cmd := exec.Command(args[0], args[1:]...)
	cmd.Dir = projectDir
	cmd.Env = append(os.Environ(), "HIVE_SESSION="+name)
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}

	ws := &pty.Winsize{Rows: uint16(height), Cols: uint16(width)}
	ptmx, err := pty.StartWithSize(cmd, ws)
	if err != nil {
		return nil, fmt.Errorf("start pty: %w", err)
	}

	sess := &Session{
		Name:       name,
		ProjectDir: projectDir,
		Cmd:        cmd,
		PTY:        ptmx,
		PID:        cmd.Process.Pid,
		ExitCh:     make(chan struct{}),
	}

	m.sessions[name] = sess

	go func() {
		cmd.Wait()
		close(sess.ExitCh)
	}()

	return sess, nil
}

func (m *Manager) Kill(name string) error {
	m.mu.Lock()
	sess, ok := m.sessions[name]
	if !ok {
		m.mu.Unlock()
		return fmt.Errorf("session %q not found", name)
	}
	delete(m.sessions, name)
	m.mu.Unlock()

	// Send SIGTERM to process group
	if sess.Cmd.Process != nil {
		syscall.Kill(-sess.PID, syscall.SIGTERM)
	}

	// Wait up to 5s for graceful exit
	select {
	case <-sess.ExitCh:
	case <-time.After(5 * time.Second):
		if sess.Cmd.Process != nil {
			syscall.Kill(-sess.PID, syscall.SIGKILL)
		}
		<-sess.ExitCh
	}

	sess.PTY.Close()
	return nil
}

func (m *Manager) Get(name string) *Session {
	m.mu.Lock()
	defer m.mu.Unlock()
	return m.sessions[name]
}

func (m *Manager) List() []*Session {
	m.mu.Lock()
	defer m.mu.Unlock()
	result := make([]*Session, 0, len(m.sessions))
	for _, s := range m.sessions {
		result = append(result, s)
	}
	return result
}

func (m *Manager) Resize(name string, width, height int) error {
	m.mu.Lock()
	sess, ok := m.sessions[name]
	m.mu.Unlock()
	if !ok {
		return fmt.Errorf("session %q not found", name)
	}
	return pty.Setsize(sess.PTY, &pty.Winsize{
		Rows: uint16(height),
		Cols: uint16(width),
	})
}

func (m *Manager) KillAll() {
	m.mu.Lock()
	names := make([]string, 0, len(m.sessions))
	for name := range m.sessions {
		names = append(names, name)
	}
	m.mu.Unlock()

	for _, name := range names {
		m.Kill(name)
	}
}

func (m *Manager) Remove(name string) {
	m.mu.Lock()
	defer m.mu.Unlock()
	delete(m.sessions, name)
}
```

```go
func EncodeProjectPath(projectPath string) string {
	resolved, err := filepath.Abs(projectPath)
	if err != nil {
		resolved = projectPath
	}
	return strings.ReplaceAll(resolved, "/", "-")

	return baseDir

func ExtractText(content interface{}) string {
	switch v := content.(type) {
	case string:
		return strings.TrimSpace(v)
	case []interface{}:
		for _, item := range v {
			m, ok := item.(map[string]interface{})
			if !ok {
				continue
			}
			if m["type"] == "text" {
				text, _ := m["text"].(string)
				if strings.TrimSpace(text) != "" {
					return strings.TrimSpace(text)
				}
			}
		}
	}
	return ""
}

func ExtractConversation(claudeProjectsBase, projectPath, sessionID string, maxChars int) []Message {
	encoded := EncodeProjectPath(projectPath)
	jsonlPath := filepath.Join(claudeProjectsBase, encoded, sessionID+".jsonl")

	f, err := os.Open(jsonlPath)
	if err != nil {
		return nil
	}
	defer f.Close()

	var allMsgs []Message
	scanner := bufio.NewScanner(f)
	scanner.Buffer(make([]byte, 256*1024), 256*1024)
	for scanner.Scan() {
		var d map[string]interface{}
		if err := json.Unmarshal(scanner.Bytes(), &d); err != nil {
			continue
		}
		msgType, _ := d["type"].(string)
		if msgType != "user" && msgType != "assistant" {
			continue
		}
		msg, _ := d["message"].(map[string]interface{})
		if msg == nil {
			continue
		}
		text := ExtractText(msg["content"])
		if text != "" {
			allMsgs = append(allMsgs, Message{Role: msgType, Text: text})
		}
	}

	if len(allMsgs) == 0 {
		return nil
	}

	// First message + last 19
	var selected []Message
	selected = append(selected, allMsgs[0])
	tail := allMsgs[1:]
	if len(tail) > 19 {
		tail = tail[len(tail)-19:]
	}
	selected = append(selected, tail...)

	// Truncate to maxChars
	total := 0
	var result []Message
	for _, m := range selected {
		remaining := maxChars - total
		if remaining <= 0 {
			break
		}
		text := m.Text
		if len(text) > remaining {
			text = text[:remaining]
		}
		result = append(result, Message{Role: m.Role, Text: text})
		total += len(text)
	}
	return result
}

func GenerateSynopsis(claudeProjectsBase, cacheDir, sessionName, projectPath, sessionID string) (string, error) {
	encoded := EncodeProjectPath(projectPath)
	jsonlPath := filepath.Join(claudeProjectsBase, encoded, sessionID+".jsonl")
	stat, err := os.Stat(jsonlPath)
	if err != nil {
		return "(no conversation yet)", nil
	}
	mtime := float64(stat.ModTime().UnixMilli()) / 1000.0
	size := int(stat.Size())

	cached := LoadCachedSynopsis(cacheDir, sessionName, mtime, size)
	if cached != "" {
		return cached, nil
	}

	msgs := ExtractConversation(claudeProjectsBase, projectPath, sessionID, 4000)
	if len(msgs) == 0 {
		return "(no conversation yet)", nil
	}

	// Try Anthropic API — if SDK not available or no key, fall back
	result, err := callAnthropicAPI(msgs)
	if err != nil {
		return FallbackSynopsis(claudeProjectsBase, projectPath, sessionID), nil
	}

	SaveCachedSynopsis(cacheDir, sessionName, sessionID, mtime, size, result)
	return result, nil
}

func callAnthropicAPI(msgs []Message) (string, error) {
	// Check for API credentials in priority order:
	// 1. CLAUDE_CODE_USE_VERTEX=1 → Vertex AI
	// 2. ANTHROPIC_FOUNDRY_API_KEY → Foundry
	// 3. ANTHROPIC_API_KEY → Direct API
	apiKey := os.Getenv("ANTHROPIC_API_KEY")
	if apiKey == "" {
		return "", fmt.Errorf("no API key available")
	}

	var conversation strings.Builder
	for _, m := range msgs {
		fmt.Fprintf(&conversation, "[%s]: %s\n", m.Role, m.Text)
	}

	// Use anthropic-sdk-go
	client := anthropic.NewClient()
	message, err := client.Messages.New(context.Background(), anthropic.MessageNewParams{
		Model:     anthropic.F("claude-haiku-4-5-20251001"),
		MaxTokens: anthropic.Int(256),
		System: []anthropic.TextBlockParam{
			{Text: anthropic.F(SystemPrompt)},
		},
		Messages: anthropic.F([]anthropic.MessageParam{
			anthropic.NewUserMessage(anthropic.NewTextBlock(conversation.String())),
		}),
	})
	if err != nil {
		return "", err
	}

	for _, block := range message.Content {
		if text := block.AsText(); text.Text != "" {
			return strings.TrimSpace(text.Text), nil
		}
	}
	return "", fmt.Errorf("no text in response")
}
```

```go
	for _, s := range sessions {
		if s.State != "bootstrapping" {
			sorted = append(sorted, s)
		}
	}
	sort.Slice(sorted, func(i, j int) bool {
		return sorted[i].Name < sorted[j].Name
	})
	m.sessionList.SetSessions(sorted)
	return m
}

func (m DashboardModel) View() string {
	headerText := m.buildHeader()
	header := HeaderStyle.Width(m.width).Render(headerText)

	footerText := "n:new  f:free  g:clone  k:kill  R:rename  u:url  ↵:attach  Q:quit"
	footer := FooterStyle.Width(m.width).Render(footerText)

	contentHeight := m.height - 2 // header + footer
	if contentHeight < 1 {
		contentHeight = 1
	}

	listWidth := m.width / 3
	if listWidth < 30 {
		listWidth = 30
	}
	previewWidth := m.width - listWidth - 1

	listContent := m.sessionList.View(listWidth, contentHeight)
	listPanel := lipgloss.NewStyle().
		Width(listWidth).

		Render(listContent)

		Height(contentHeight).

	body := lipgloss.JoinHorizontal(lipgloss.Top, listPanel, separator, previewPanel)

func (m DashboardModel) buildHeader() string {

	title := "Claude Code Orchestrator"

	if m.startingUp {
		return fmt.Sprintf("%s  │  %s  │  starting up…", brand, title)
	}

	booting := 0
	waiting := 0
	ready := 0
	for _, s := range m.sessions {
		switch s.State {
		case "bootstrapping":
			booting++
		case "waiting":
			waiting++
			ready++
		default:
			ready++
		}
	}

	parts := []string{brand, title, fmt.Sprintf("%d sessions", ready)}
	if waiting > 0 {
		parts = append(parts, BadgeWaitingStyle.Render(fmt.Sprintf(" %d Waiting ", waiting)))
	}
	if booting > 0 {
		parts = append(parts, BadgeLoadingStyle.Render(fmt.Sprintf(" %d Loading ", booting)))
	}
	return strings.Join(parts, "  │  ")
}

	}

}
```

```go
package ui

import (
	"fmt"
	"net"

	ModeDashboard Mode = iota
	ModeFocus
	ModeDialog

	dialogNone dialogKind = iota
	dialogProjectPicker
	dialogSessionName
	dialogConfirmKill
	dialogRename
	dialogClone
	dialogError

// Messages for async operations
type pollResultMsg struct {
	sessions map[string]SessionData
}

type synopsisResultMsg struct {
	name     string
	synopsis string
}

type AppModel struct {
	config     config.HiveConfig
	state      *config.HiveState
	procMgr    *process.Manager
	dashboard  DashboardModel
	terminal   *TerminalWidget
	focusedName string

	mode       Mode
	dialog     dialogKind
	projectPicker ProjectPickerModel
	sessionName   SessionNameModel
	confirmKill   ConfirmKillModel
	rename        RenameModel
	clone         CloneModel
	errorDialog   ErrorModel

	pendingProject ProjectInfo

	width  int
	height int

	hookStateDir   string
	synopsisCacheDir string
	claudeProjectsDir string
}

func NewApp(cfg config.HiveConfig, state *config.HiveState) AppModel {
	home, _ := os.UserHomeDir()
	return AppModel{
		config:        cfg,
		state:         state,
		procMgr:       process.NewManager(),
		mode:          ModeDashboard,
		hookStateDir:  filepath.Join(home, ".claude", "hive", "state"),
		synopsisCacheDir: filepath.Join(home, ".claude", "hive", "synopsis"),
		claudeProjectsDir: filepath.Join(home, ".claude", "projects"),
	}
}

func (m AppModel) Init() tea.Cmd {
	return tea.Batch(
		tea.EnterAltScreen,
		m.pollCmd(),
	)
}

func (m AppModel) pollCmd() tea.Cmd {
	return tea.Tick(time.Duration(m.config.RefreshIntervalMs)*time.Millisecond, func(t time.Time) tea.Msg {
		sessions := m.buildSessionData()
		return pollResultMsg{sessions: sessions}
	})
}

func (m AppModel) buildSessionData() map[string]SessionData {
	sessions := make(map[string]SessionData)
	for _, sess := range m.procMgr.List() {
		hookState, hookUpdatedAt := hookstate.ReadWithMeta(sess.Name, m.hookStateDir)
		state := hookState
		if state == "" {
			state = detector.StateWorking
		}
		waitingSince := ""
		if state == detector.StateWaiting {
			waitingSince = hookUpdatedAt
		}

		sessionID := hookstate.ReadSessionID(sess.Name, m.hookStateDir)
		projectPath := ""
		if info, ok := m.state.Sessions[sess.Name]; ok {
			projectPath = info.ProjectPath
		} else {
			projectPath = sess.ProjectDir
		}

		sd := SessionData{
			Name:         sess.Name,
			ProjectPath:  projectPath,
			State:        state,
			ContextPct:   -1,
			WaitingSince: waitingSince,
		}

		if sessionID != "" {
			cached := synopsis.LoadCachedSynopsis(m.synopsisCacheDir, sess.Name, 0, 0)
			if cached != "" {
				sd.Synopsis = cached
			}
		}

		sessions[sess.Name] = sd
	}
	return sessions
}

func (m AppModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width = msg.Width
		m.height = msg.Height
		m.dashboard.width = msg.Width
		m.dashboard.height = msg.Height
		if m.terminal != nil {
			tw, cmd := m.terminal.Update(msg)
			m.terminal = tw
			return m, cmd
		}

	case pollResultMsg:
		m.dashboard = m.dashboard.UpdateSessions(msg.sessions)
		return m, m.pollCmd()

	case synopsisResultMsg:
		// Update synopsis in session data
		return m, nil

	case DialogCancelMsg:
		m.mode = ModeDashboard
		m.dialog = dialogNone
		return m, nil

	case DialogResultMsg:
		return m.handleDialogResult(msg)
	}

	switch m.mode {
	case ModeFocus:
		return m.updateFocus(msg)
	case ModeDialog:
		return m.updateDialog(msg)
	default:
		return m.updateDashboard(msg)
	}
}

func (m AppModel) updateDashboard(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyMsg:
		switch msg.String() {
		case "Q":
			m.procMgr.KillAll()
			return m, tea.Quit
		case "n":
			projects := m.scanProjects()
			m.projectPicker = NewProjectPicker(projects, m.width, m.height)
			m.mode = ModeDialog
			m.dialog = dialogProjectPicker
			return m, nil
		case "f":
			home, _ := os.UserHomeDir()
			m.pendingProject = ProjectInfo{Name: "home", Path: home}
			m.sessionName = NewSessionName("home (~)")
			m.mode = ModeDialog
			m.dialog = dialogSessionName
			return m, nil
		case "g":
			m.clone = NewClone(m.config.ClonePath)
			m.mode = ModeDialog
			m.dialog = dialogClone
			return m, nil
		case "k":
			if sel := m.dashboard.SelectedSession(); sel != nil {
				m.confirmKill = NewConfirmKill(sel.Name)
				m.mode = ModeDialog
				m.dialog = dialogConfirmKill
				return m, nil
			}
		case "r":
			if sel := m.dashboard.SelectedSession(); sel != nil {
				return m.resumeSession(sel.Name)
			}
		case "R":
			if sel := m.dashboard.SelectedSession(); sel != nil {
				m.rename = NewRename(sel.Name)
				m.mode = ModeDialog
				m.dialog = dialogRename
				return m, nil
			}
		case "u":
			if sel := m.dashboard.SelectedSession(); sel != nil && len(sel.URLs) > 0 {
				// Open first URL
				// webbrowser.open equivalent — use xdg-open
				return m, nil
			}
		case "enter":
			if sel := m.dashboard.SelectedSession(); sel != nil {
				return m.enterFocus(sel.Name)
			}
		}
	}

	m.dashboard, _ = m.dashboard.Update(msg)
	return m, nil
}

func (m AppModel) updateFocus(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyMsg:
		if msg.Type == tea.KeyF1 {
			if m.terminal != nil {
				m.terminal.Blur()
			}
			m.mode = ModeDashboard
			return m, nil
		}
	}

	if m.terminal != nil {
		tw, cmd := m.terminal.Update(msg)
		m.terminal = tw
		return m, cmd
	}
	return m, nil
}

func (m AppModel) updateDialog(msg tea.Msg) (tea.Model, tea.Cmd) {
	var cmd tea.Cmd
	switch m.dialog {

func (m AppModel) createSession(projectPath, name, resumeID string) (tea.Model, tea.Cmd) {

func (m AppModel) scanProjects() []ProjectInfo {
	projectMap := make(map[string]ProjectInfo)
	for _, scanPath := range m.config.ScanPaths {
		expanded := os.ExpandEnv(scanPath)
		if strings.HasPrefix(expanded, "~") {
			home, _ := os.UserHomeDir()
			expanded = filepath.Join(home, expanded[1:])
		}
		entries, err := os.ReadDir(expanded)
		if err != nil {
			continue
		}
		for _, entry := range entries {
			if !entry.IsDir() || strings.HasPrefix(entry.Name(), ".") {
				continue
			}
			full := filepath.Join(expanded, entry.Name())
			if _, exists := projectMap[full]; exists {
				continue
			}
			age := ""
			if pi, ok := m.state.Projects[full]; ok {
				age = formatAge(pi.LastUsed)
			}
			projectMap[full] = ProjectInfo{
				Name: entry.Name(),
				Path: full,
				Age:  age,
			}
		}
	}

	var projects []ProjectInfo
	for _, p := range projectMap {
		projects = append(projects, p)
	}
	sort.Slice(projects, func(i, j int) bool {
		return projects[i].Name < projects[j].Name
	})
	return projects
}

func (m AppModel) nextSessionName(baseName string) string {
	existing := make(map[string]bool)
	for _, sess := range m.procMgr.List() {
		existing[sess.Name] = true
	}
	i := 1
	for {
		name := fmt.Sprintf("%s-%03d", baseName, i)
		if !existing[name] {
			return name
		}
		i++
	}
}

func formatAge(isoStr string) string {
	if isoStr == "" {
		return ""
	}
	t, err := time.Parse(time.RFC3339, isoStr)
	if err != nil {
		return ""
	}
	diff := time.Since(t)
	days := int(diff.Hours() / 24)
	if days > 0 {
		return fmt.Sprintf("%dd", days)
	}
	hours := int(diff.Hours())
	if hours > 0 {
		return fmt.Sprintf("%dh", hours)
	}
	return fmt.Sprintf("%dm", int(diff.Minutes()))
}

	}
	conn.Close()
	return true
}
```

```go
type restoreCompleteMsg struct {
	spawned []string

func (m AppModel) restoreSessionsCmd() tea.Cmd {
	return func() tea.Msg {
		var spawned []string
		var errors []string

		for name, info := range m.state.Sessions {
			if err := safety.ValidateSessionName(name); err != nil {
				m.state.RemoveSession(name)
				continue
			}
			if !dirExists(info.ProjectPath) {
				m.state.RemoveSession(name)
				continue
			}

			// Check if process is still alive
			if info.PID > 0 && processAlive(info.PID) {
				continue
			}

			// Re-spawn
			hookstate.Remove(name, m.hookStateDir)
			args := []string{"claude", "--dangerously-skip-permissions", "--name", name}
			if info.ClaudeSessionID != "" {
				args = append(args, "--resume", info.ClaudeSessionID)
			} else {
				args = append(args, "--continue")
			}

			sess, err := m.procMgr.Spawn(name, info.ProjectPath, args, 200, 50)
			if err != nil {
				errors = append(errors, fmt.Sprintf("%s: %v", name, err))
				continue
			}
			si := m.state.Sessions[name]
			si.PID = sess.PID
			m.state.Sessions[name] = si
			spawned = append(spawned, name)
		}
		m.state.SaveDefault()
		return restoreCompleteMsg{spawned: spawned, errors: errors}
	}
}

func dirExists(path string) bool {
	info, err := os.Stat(path)
	return err == nil && info.IsDir()
}

func processAlive(pid int) bool {
	proc, err := os.FindProcess(pid)
	if err != nil {
		return false
	}
	// On Unix, FindProcess always succeeds. Send signal 0 to check.
	err = proc.Signal(syscall.Signal(0))
	return err == nil
}
```
