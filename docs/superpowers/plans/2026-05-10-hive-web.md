# Hive Web Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Textual + tmux dashboard in `src/hive/` with a FastAPI + xterm.js browser multiplexer that derives session state from Claude Code's JSONL transcripts (no TUI parsing) and owns the PTYs of spawned `claude` processes (no tmux).

**Architecture:** Python FastAPI server with three single-responsibility modules — `SessionManager` (spawn/kill `claude` subprocesses in PTYs), `StateWatcher` (watch `~/.claude/projects/` and derive state from JSONL events), `PtyBridge` (pump bytes between WebSockets and PTY fds). Frontend is a single HTML page with vanilla JS + xterm.js. Spec: `docs/superpowers/specs/2026-05-10-hive-web-design.md`.

**Tech Stack:** Python 3.11+, FastAPI, uvicorn[standard], watchdog, stdlib `pty`/`subprocess`/`asyncio`. Frontend: vanilla JS, xterm.js (vendored from CDN). Tests: pytest, pytest-asyncio, FastAPI TestClient.

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from pathlib import Path


class SessionState(Enum):
    BOOTSTRAPPING = "bootstrapping"
    WORKING = "working"
    WAITING = "waiting"
    DEAD = "dead"


@dataclass
class Session:
    id: str
    name: str
    project_path: Path
    created_at: datetime
    pid: int | None = None
    pty_master_fd: int | None = None
    jsonl_path: Path | None = None
    claude_session_id: str | None = None
    state: SessionState = SessionState.BOOTSTRAPPING

    def to_persisted(self) -> dict:
        return {
            "id": self.id,
            "name": self.name,
            "project_path": str(self.project_path),
            "claude_session_id": self.claude_session_id,
        }
```

```python
from __future__ import annotations

import json
from datetime import datetime, timedelta
from pathlib import Path

from hive.session import SessionState

CONVERSATION_TYPES = {"user", "assistant"}
STALE_AFTER = timedelta(minutes=5)
TAIL_CHUNK_BYTES = 8192


def read_last_conversation_event(path: Path) -> dict | None:
    """Read the file backward in 8KB chunks; return last user/assistant event."""
    try:
        size = path.stat().st_size
    except FileNotFoundError:
        return None
    if size == 0:
        return None

    buf = b""
    pos = size
    with path.open("rb") as f:
        while pos > 0:
            read_size = min(TAIL_CHUNK_BYTES, pos)
            pos -= read_size
            f.seek(pos)
            buf = f.read(read_size) + buf
            lines = buf.split(b"\n")
            buf = lines[0]  # keep partial line for next iteration
            for line in reversed(lines[1:]):
                line = line.strip()
                if not line:
                    continue
                try:
                    ev = json.loads(line)
                except json.JSONDecodeError:
                    continue
                if ev.get("type") in CONVERSATION_TYPES:
                    return ev
        # Process any remaining first-chunk content
        line = buf.strip()
        if line:
            try:
                ev = json.loads(line)
                if ev.get("type") in CONVERSATION_TYPES:
                    return ev
            except json.JSONDecodeError:
                pass
    return None


def derive_state(jsonl_path: Path, now: datetime) -> SessionState:
    last = read_last_conversation_event(jsonl_path)
    if last is None:
        return SessionState.BOOTSTRAPPING

    ts = datetime.fromisoformat(last["timestamp"].replace("Z", "+00:00"))
    if now - ts > STALE_AFTER:
        return SessionState.WAITING

    msg = last.get("message") or {}
    stop = msg.get("stop_reason")
    if last["type"] == "assistant":
        if stop == "end_turn":
            return SessionState.WAITING
        return SessionState.WORKING
    return SessionState.WORKING  # type == "user"
```

```python
CONTEXT_WINDOWS = {
    "claude-opus-4-7": 200_000,
    "claude-opus-4-6": 200_000,
    "claude-sonnet-4-6": 200_000,
    "claude-haiku-4-5": 200_000,
}


def derive_context_pct(jsonl_path: Path, model: str | None) -> int | None:
    """Walk JSONL; find latest assistant message with `usage`; compute pct."""
    if model is None or model not in CONTEXT_WINDOWS:
        return None
    last_usage = None
    try:
        with jsonl_path.open() as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                try:
                    ev = json.loads(line)
                except json.JSONDecodeError:
                    continue
                if ev.get("type") != "assistant":
                    continue
                usage = (ev.get("message") or {}).get("usage")
                if usage:
                    last_usage = usage
    except OSError:
        return None
    if not last_usage:
        return None
    total = (
        last_usage.get("input_tokens", 0)
        + last_usage.get("cache_creation_input_tokens", 0)
        + last_usage.get("cache_read_input_tokens", 0)
        + last_usage.get("output_tokens", 0)
    )
    window = CONTEXT_WINDOWS[model]
    return min(round(total / window * 100), 100)
```

```python
from __future__ import annotations

import json
import os
import tempfile
from pathlib import Path

from hive.session import Session


class StatePersistence:
    def __init__(self, path: Path):
        self.path = path

    def load(self) -> list[dict]:
        try:
            with self.path.open() as f:
                data = json.load(f)
        except (FileNotFoundError, json.JSONDecodeError, OSError):
            return []
        sessions = data.get("sessions")
        if not isinstance(sessions, list):
            return []
        return sessions

    def save(self, sessions: list[Session]) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        payload = {"sessions": [s.to_persisted() for s in sessions]}
        # Atomic write: write to temp file in same dir, then rename.
        fd, tmp_name = tempfile.mkstemp(
            dir=self.path.parent, prefix=self.path.name + ".tmp."
        )
        try:
            with os.fdopen(fd, "w") as f:
                json.dump(payload, f, indent=2)
            os.replace(tmp_name, self.path)
        except Exception:
            try:
                os.unlink(tmp_name)
            except OSError:
                pass
            raise
```

```python
from __future__ import annotations

import os
import pty
import subprocess
import uuid
from datetime import datetime, timezone
from pathlib import Path

from hive.session import Session, SessionState


class SessionManager:
    def __init__(
        self,
        claude_command: list[str] | None = None,
        claude_projects_dir: Path | None = None,
        state_path: Path | None = None,
    ):
        self.claude_command = claude_command or ["claude"]
        self.claude_projects_dir = (
            claude_projects_dir
            or Path.home() / ".claude" / "projects"
        )
        self.state_path = (
            state_path
            or Path.home() / ".config" / "hive" / "state.json"
        )
        self._sessions: dict[str, Session] = {}

    def spawn(self, project_path: Path, name: str) -> Session:
        if any(s.name == name for s in self._sessions.values()):
            raise ValueError(f"session name '{name}' already exists")

        master_fd, slave_fd = pty.openpty()
        env = {
            **os.environ,
            "TERM": "xterm-256color",
        }
        cmd = list(self.claude_command) + [
            "--dangerously-skip-permissions",
            "--name", name,
        ]
        # Only pass --projects-dir to the spawned binary if we're overriding
        # the default. Real `claude` doesn't accept this flag; fake_claude
        # (used in tests) does.
        default_dir = Path.home() / ".claude" / "projects"
        if self.claude_projects_dir != default_dir:
            cmd += ["--projects-dir", str(self.claude_projects_dir)]
        try:
            proc = subprocess.Popen(
                cmd,
                cwd=str(project_path),
                env=env,
                stdin=slave_fd,
                stdout=slave_fd,
                stderr=slave_fd,
                start_new_session=True,
                close_fds=True,
            )
        finally:
            os.close(slave_fd)

        session = Session(
            id=str(uuid.uuid4()),
            name=name,
            project_path=project_path,
            created_at=datetime.now(timezone.utc),
            pid=proc.pid,
            pty_master_fd=master_fd,
            state=SessionState.BOOTSTRAPPING,
        )
        self._sessions[session.id] = session
        return session
```

Polling-based to avoid platform-specific surprises and to keep tests deterministic. Each tick, re-derive state for every bound session; emit only on change.

```python
from __future__ import annotations

import asyncio
import threading
from datetime import datetime, timezone
from pathlib import Path

from hive.derive import derive_state
from hive.session import SessionState


class StateWatcher:
    def __init__(self, poll_interval: float = 0.5):
        self.poll_interval = poll_interval
        self._bindings: dict[str, Path] = {}
        self._last_state: dict[str, SessionState] = {}
        self._subscribers: list[asyncio.Queue] = []
        self._lock = threading.Lock()
        self._loop: asyncio.AbstractEventLoop | None = None
        self._task: asyncio.Task | None = None
        self._running = False

    def bind(self, session_id: str, jsonl_path: Path) -> None:
        with self._lock:
            self._bindings[session_id] = jsonl_path
            self._last_state.pop(session_id, None)

    def unbind(self, session_id: str) -> None:
        with self._lock:
            self._bindings.pop(session_id, None)
            self._last_state.pop(session_id, None)

    def subscribe(self) -> asyncio.Queue:
        q: asyncio.Queue = asyncio.Queue()
        with self._lock:
            self._subscribers.append(q)
        return q

    def start(self) -> None:
        if self._running:
            return
        self._running = True
        self._loop = asyncio.get_event_loop()
        self._task = self._loop.create_task(self._poll_loop())

    def stop(self) -> None:
        self._running = False
        if self._task is not None:
            self._task.cancel()
            self._task = None

    async def _poll_loop(self) -> None:
        try:
            while self._running:
                await self._tick()
                await asyncio.sleep(self.poll_interval)
        except asyncio.CancelledError:
            pass

    async def _tick(self) -> None:
        with self._lock:
            snapshot = list(self._bindings.items())
        now = datetime.now(timezone.utc)
        for sid, path in snapshot:
            new_state = derive_state(path, now)
            old_state = self._last_state.get(sid)
            if new_state != old_state:
                self._last_state[sid] = new_state
                await self._broadcast((sid, new_state))

    async def _broadcast(self, evt: tuple[str, SessionState]) -> None:
        with self._lock:
            subs = list(self._subscribers)
        for q in subs:
            await q.put(evt)

    async def emit(self, session_id: str, state: SessionState) -> None:
        """Broadcast a state change without going through JSONL derivation.

        Used by the reaper to publish DEAD events when a process exits.
        """
        self._last_state[session_id] = state
        await self._broadcast((session_id, state))
```

```python
from __future__ import annotations

import asyncio
import os
from collections import deque


class _PtyState:
    def __init__(self, fd: int, scrollback_bytes: int):
        self.fd = fd
        self.scrollback = deque()
        self.scrollback_size = 0
        self.scrollback_max = scrollback_bytes
        self.viewers: set = set()  # set of asyncio.Queue


class PtyBridge:
    def __init__(self, scrollback_bytes: int = 256 * 1024):
        self.scrollback_bytes = scrollback_bytes
        self._ptys: dict[str, _PtyState] = {}

    def register_pty(self, session_id: str, master_fd: int) -> None:
        state = _PtyState(master_fd, self.scrollback_bytes)
        self._ptys[session_id] = state
        loop = asyncio.get_event_loop()
        loop.add_reader(master_fd, self._on_pty_readable, session_id)

    def unregister_pty(self, session_id: str) -> None:
        state = self._ptys.pop(session_id, None)
        if state is None:
            return
        loop = asyncio.get_event_loop()
        try:
            loop.remove_reader(state.fd)
        except (ValueError, OSError):
            pass

    def _on_pty_readable(self, session_id: str) -> None:
        state = self._ptys.get(session_id)
        if state is None:
            return
        try:
            data = os.read(state.fd, 4096)
        except OSError:
            return
        if not data:
            return
        # Append to scrollback (trim to max)
        state.scrollback.append(data)
        state.scrollback_size += len(data)
        while state.scrollback_size > state.scrollback_max and state.scrollback:
            old = state.scrollback.popleft()
            state.scrollback_size -= len(old)
        # Fan out to viewers
        for q in list(state.viewers):
            try:
                q.put_nowait(data)
            except asyncio.QueueFull:
                pass

    async def attach(self, session_id: str, websocket) -> None:
        state = self._ptys.get(session_id)
        if state is None:
            return

        viewer_q: asyncio.Queue = asyncio.Queue(maxsize=1024)
        state.viewers.add(viewer_q)

        # Replay scrollback
        for chunk in list(state.scrollback):
            await websocket.send_bytes(chunk)

        async def pty_to_socket():
            while True:
                data = await viewer_q.get()
                await websocket.send_bytes(data)

        async def socket_to_pty():
            while True:
                data = await websocket.receive_bytes()
                if state.fd is not None:
                    try:
                        os.write(state.fd, data)
                    except OSError:
                        return

        out_task = asyncio.create_task(pty_to_socket())
        in_task = asyncio.create_task(socket_to_pty())
        try:
            done, pending = await asyncio.wait(
                [out_task, in_task],
                return_when=asyncio.FIRST_COMPLETED,
            )
            for t in pending:
                t.cancel()
        finally:
            state.viewers.discard(viewer_q)
```

```python
from __future__ import annotations

from pathlib import Path

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel

from hive.manager import SessionManager
from hive.session import Session
from hive.watcher import StateWatcher


class CreateSessionRequest(BaseModel):
    project_path: str
    name: str


def session_to_dict(s: Session) -> dict:
    return {
        "id": s.id,
        "name": s.name,
        "project_path": str(s.project_path),
        "state": s.state.value,
        "claude_session_id": s.claude_session_id,
        "created_at": s.created_at.isoformat(),
    }


def build_router(manager: SessionManager, watcher: StateWatcher) -> APIRouter:
    router = APIRouter()

    @router.get("/sessions")
    def list_sessions():
        return {"sessions": [session_to_dict(s) for s in manager.list()]}

    @router.post("/sessions", status_code=201)
    def create_session(req: CreateSessionRequest):
        project = Path(req.project_path)
        if not project.is_dir():
            raise HTTPException(
                status_code=400, detail=f"not a directory: {project}"
            )
        try:
            session = manager.spawn(project_path=project, name=req.name)
        except ValueError as e:
            raise HTTPException(status_code=409, detail=str(e))
        manager.save_state()
        return session_to_dict(session)

    @router.delete("/sessions/{session_id}", status_code=204)
    def delete_session(session_id: str):
        s = manager.get(session_id)
        if s is None:
            raise HTTPException(status_code=404, detail="unknown session")
        watcher.unbind(session_id)
        manager.kill(session_id)
        manager.save_state()
        return None

    @router.get("/healthz")
    def healthz():
        return {"sessions": len(manager.list()), "ok": True}

    return router
```

```python
from __future__ import annotations

import asyncio
import logging
import os
from contextlib import asynccontextmanager
from pathlib import Path

from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from fastapi.staticfiles import StaticFiles

from hive.manager import SessionManager
from hive.pty_bridge import PtyBridge
from hive.routes import build_router
from hive.session import SessionState
from hive.watcher import StateWatcher

log = logging.getLogger(__name__)

JSONL_DISCOVERY_TIMEOUT = 30.0
JSONL_DISCOVERY_POLL = 0.5


def _encode_project_path(p: Path) -> str:
    return str(p).replace("/", "-")


async def _discover_and_bind_jsonl(
    manager: SessionManager,
    watcher: StateWatcher,
    bridge: PtyBridge,
    session_id: str,
) -> None:
    s = manager.get(session_id)
    if s is None:
        return
    bridge.register_pty(session_id, s.pty_master_fd)

    spawn_time = s.created_at.timestamp()
    encoded_dir = manager.claude_projects_dir / _encode_project_path(s.project_path)
    deadline = asyncio.get_event_loop().time() + JSONL_DISCOVERY_TIMEOUT
    while asyncio.get_event_loop().time() < deadline:
        if encoded_dir.exists():
            candidates = sorted(
                (p for p in encoded_dir.glob("*.jsonl")
                 if p.stat().st_mtime >= spawn_time - 1),
                key=lambda p: p.stat().st_mtime,
                reverse=True,
            )
            if candidates:
                jsonl = candidates[0]
                s.jsonl_path = jsonl
                s.claude_session_id = jsonl.stem
                manager.save_state()
                watcher.bind(session_id, jsonl)
                return
        await asyncio.sleep(JSONL_DISCOVERY_POLL)
    log.error("session %s: jsonl never appeared", session_id)


REAPER_INTERVAL = 1.0


async def _reap_dead_sessions(
    manager: SessionManager,
    watcher: StateWatcher,
    bridge: PtyBridge,
) -> None:
    """Periodically check spawned processes; mark crashed ones DEAD."""
    while True:
        await asyncio.sleep(REAPER_INTERVAL)
        for s in list(manager.list()):
            if s.pid is None or s.state == SessionState.DEAD:
                continue
            try:
                os.kill(s.pid, 0)
            except (ProcessLookupError, PermissionError):
                s.state = SessionState.DEAD
                s.pid = None
                if s.pty_master_fd is not None:
                    try:
                        os.close(s.pty_master_fd)
                    except OSError:
                        pass
                    s.pty_master_fd = None
                bridge.unregister_pty(s.id)
                watcher.unbind(s.id)
                await watcher.emit(s.id, SessionState.DEAD)


def build_app(
    claude_command: list[str] | None = None,
    claude_projects_dir: Path | None = None,
    state_path: Path | None = None,
    static_dir: Path | None = None,
) -> FastAPI:
    manager = SessionManager(
        claude_command=claude_command,
        claude_projects_dir=claude_projects_dir,
        state_path=state_path,
    )
    watcher = StateWatcher(poll_interval=0.5)
    bridge = PtyBridge()
    discovery_started: set[str] = set()

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        watcher.start()
        manager.restart_persisted()
        for s in manager.list():
            discovery_started.add(s.id)
            asyncio.create_task(
                _discover_and_bind_jsonl(manager, watcher, bridge, s.id)
            )
        reaper_task = asyncio.create_task(
            _reap_dead_sessions(manager, watcher, bridge)
        )
        yield
        reaper_task.cancel()
        for s in manager.list():
            manager.kill(s.id)
        watcher.stop()

    app = FastAPI(lifespan=lifespan)
    app.state.manager = manager
    app.state.watcher = watcher
    app.state.bridge = bridge
    app.include_router(build_router(manager, watcher))

    @app.middleware("http")
    async def post_spawn_discovery(request, call_next):
        response = await call_next(request)
        for s in manager.list():
            if (
                s.jsonl_path is None
                and s.pty_master_fd is not None
                and s.id not in discovery_started
            ):
                discovery_started.add(s.id)
                asyncio.create_task(
                    _discover_and_bind_jsonl(manager, watcher, bridge, s.id)
                )
        return response

    @app.websocket("/ws/state")
    async def ws_state(ws: WebSocket):
        await ws.accept()
        queue = watcher.subscribe()
        # Send current state for all known sessions on connect
        for s in manager.list():
            await ws.send_json({"session_id": s.id, "state": s.state.value})
        try:
            while True:
                sid, state = await queue.get()
                # Update in-memory session state
                s = manager.get(sid)
                if s is not None:
                    s.state = state
                await ws.send_json({"session_id": sid, "state": state.value})
        except WebSocketDisconnect:
            return

    @app.websocket("/ws/pty/{session_id}")
    async def ws_pty(ws: WebSocket, session_id: str):
        await ws.accept()
        # Adapter so PtyBridge.attach gets the receive_bytes/send_bytes API
        class _Adapter:
            async def receive_bytes(self):
                return await ws.receive_bytes()
            async def send_bytes(self, data: bytes):
                await ws.send_bytes(data)
            async def close(self):
                await ws.close()
        try:
            await bridge.attach(session_id, _Adapter())
        except WebSocketDisconnect:
            return

    if static_dir is not None and static_dir.is_dir():
        app.mount("/", StaticFiles(directory=str(static_dir), html=True), name="static")

    return app
```

(Note: the middleware-based discovery hook is a deliberate simplification so we don't have to refactor the routes contract. A cleaner design — refactoring `routes.create_session` to receive a callback — can come in a follow-up.)

```javascript
const sessionList = document.getElementById('session-list');
const terminalHost = document.getElementById('terminal-host');
const terminalEmpty = document.getElementById('terminal-empty');
const newBtn = document.getElementById('new-session-btn');
const dialog = document.getElementById('new-session-dialog');
const dialogForm = dialog.querySelector('form');

const STATE_ICON = {
  bootstrapping: '◌',
  working: '◐',
  waiting: '●',
  dead: '✕',
};

const sessions = new Map();   // id -> {id, name, state, ...}
let activeSessionId = null;
let activeTerm = null;
let activeFit = null;
let activePtySocket = null;

function renderList() {
  sessionList.innerHTML = '';
  for (const s of sessions.values()) {
    const li = document.createElement('li');
    li.className = `state-${s.state}` + (s.id === activeSessionId ? ' active' : '');
    li.dataset.id = s.id;
    li.innerHTML = `
      <span class="icon">${STATE_ICON[s.state] ?? '?'}</span>
      <span class="name">${s.name}</span>
      <button class="kill" title="Kill">✕</button>
    `;
    li.querySelector('.kill').addEventListener('click', (e) => {
      e.stopPropagation();
      killSession(s.id);
    });
    li.addEventListener('click', () => attachToSession(s.id));
    sessionList.appendChild(li);
  }
}

async function killSession(id) {
  if (!confirm(`Kill ${sessions.get(id)?.name ?? id}?`)) return;
  await fetch(`/sessions/${id}`, { method: 'DELETE' });
  sessions.delete(id);
  if (activeSessionId === id) detachTerminal();
  renderList();
}

function detachTerminal() {
  if (activePtySocket) { activePtySocket.close(); activePtySocket = null; }
  if (activeTerm) { activeTerm.dispose(); activeTerm = null; activeFit = null; }
  terminalHost.replaceChildren();
  terminalHost.hidden = true;
  terminalEmpty.hidden = false;
  activeSessionId = null;
}

function attachToSession(id) {
  if (activeSessionId === id) return;
  detachTerminal();

  activeSessionId = id;
  terminalEmpty.hidden = true;
  terminalHost.hidden = false;

  activeTerm = new Terminal({
    convertEol: true,
    cursorBlink: true,
    fontSize: 13,
    fontFamily: 'monospace',
  });
  activeFit = new FitAddon.FitAddon();
  activeTerm.loadAddon(activeFit);
  activeTerm.open(terminalHost);
  activeFit.fit();

  const wsProto = location.protocol === 'https:' ? 'wss:' : 'ws:';
  const ws = new WebSocket(`${wsProto}//${location.host}/ws/pty/${id}`);
  ws.binaryType = 'arraybuffer';

  ws.onmessage = (ev) => {
    if (typeof ev.data === 'string') {
      activeTerm.write(ev.data);
    } else {
      const u8 = new Uint8Array(ev.data);
      activeTerm.write(u8);
    }
  };

  activeTerm.onData((d) => {
    ws.send(new TextEncoder().encode(d));
  });

  activePtySocket = ws;
  renderList();
}

window.addEventListener('resize', () => {
  if (activeFit) activeFit.fit();
});

newBtn.addEventListener('click', () => dialog.showModal());

dialog.addEventListener('close', async () => {
  if (dialog.returnValue !== 'ok') return;
  const fd = new FormData(dialogForm);
  const r = await fetch('/sessions', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      project_path: fd.get('project_path'),
      name: fd.get('name'),
    }),
  });
  dialogForm.reset();
  if (!r.ok) {
    alert(`Create failed: ${await r.text()}`);
    return;
  }
  const created = await r.json();
  sessions.set(created.id, created);
  renderList();
});

async function loadInitial() {
  const r = await fetch('/sessions');
  const data = await r.json();
  for (const s of data.sessions) sessions.set(s.id, s);
  renderList();
}

function connectStateSocket() {
  const wsProto = location.protocol === 'https:' ? 'wss:' : 'ws:';
  const ws = new WebSocket(`${wsProto}//${location.host}/ws/state`);
  ws.onmessage = (ev) => {
    const evt = JSON.parse(ev.data);
    const existing = sessions.get(evt.session_id);
    if (existing) {
      existing.state = evt.state;
      renderList();
    } else {
      // Session unknown to us — refetch list
      loadInitial();
    }
  };
  ws.onclose = () => setTimeout(connectStateSocket, 1000);
}

loadInitial().then(connectStateSocket);
```

# Hive

Browser multiplexer for Claude Code sessions. Replaces tmux.

## Quick start

```bash
uv sync
uv run hive
# open http://127.0.0.1:8765
```

## How it works

- Hive spawns each `claude` session in a PTY it owns; no tmux.
- The browser shows a sidebar of sessions and the active session's TUI in an `xterm.js` panel.
- Session state (working/waiting) comes from Claude Code's own JSONL transcripts in `~/.claude/projects/`, not from screen scraping. Reliable.

## CLI flags

- `--host` — bind address (default `127.0.0.1`)
- `--port` — bind port (default `8765`)
- `--state-path` — where session metadata is persisted (default `~/.config/hive/state.json`)
- `--claude-projects-dir` — Claude Code transcripts root (default `~/.claude/projects`)
- `--claude-cmd` — claude binary name or path (default `claude`)

```bash
uv run hive-legacy
```
